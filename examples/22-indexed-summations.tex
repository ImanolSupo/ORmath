\subsection*{One multidimensional sum and explicit nested sums}
\begin{ormodel}[numbering=none]
  \minimize \Sum{i in I,j in J_i} c_{ij}x_{ij} \\
  \st \Sum{i in I}\Sum{j in J_i} c_{ij}x_{ij} <= B
\end{ormodel}
\subsection*{Filtered memberships and raw mathematics}
\begin{ormodel}
  \maximize \Sum{i in \mathcal{I} where d_i >= 0} p_i x_i \\
  \st \Sum{i in I,j in J where a_{ij}=1} x_{ij} <= K \\
  \Sum*{\substack{i\in I\\j\in J_i\\(i,j)\notin A}} x_{ij} <= M
\end{ormodel}
Outside the environment, the author's command is restored: \Sum.

\subsection*{Structured expressions and namespaced command}
\begin{ormodel}
  \objective[max]{\orsum{i in I} p_i x_i}
  \constraint[for={i\in I}]{\orsum{j in J_i} a_{ij}x_j\leq b_i}
\end{ormodel}
The public namespaced operator also works in ordinary math:
\[\orsum{i in I where f(i,j)>0,\;g(i,j)<=5} x_i
  = \sum_{i\in I:\,f(i,j)>0,\;g(i,j)\leq5} x_i.\]
